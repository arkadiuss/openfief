{{- /* Every relative link is checked at build time, so a broken one fails the build: - a Markdown file or folder under docs/ or site/ becomes the site page it renders to; - any other repository path must exist and becomes a GitHub link pinned to the built commit; - a same-page #fragment must name a heading on the page; - a site path is made relative to the site's base URL; on a docs page it names a docs path, so /status/ is /docs/status/, the same in hand-written and generated pages. */ -}} {{- $dest := .Destination -}} {{- $href := $dest -}} {{- $u := urls.Parse $dest -}} {{- $p := .PageInner -}} {{- $from := "" -}} {{- with $p.File -}}{{- $from = strings.TrimPrefix (printf "%s/" hugo.WorkingDir) .Filename -}}{{- end -}} {{- if or $u.Scheme (hasPrefix $dest "//") -}} {{- /* External link: kept as written. */ -}} {{- else if hasPrefix $dest "/" -}} {{- /* Site path such as the generated requirements report: made relative to the site's base URL. */ -}} {{- $path := strings.TrimPrefix "/" $dest -}} {{- if hasPrefix $p.RelPermalink "/docs/" -}}{{- $path = printf "docs/%s" $path -}}{{- end -}} {{- $href = relURL $path -}} {{- else if hasPrefix $dest "#" -}} {{- if not ($p.Fragments.Identifiers.Contains $u.Fragment) -}} {{- errorf "%s: link %q names no heading on this page" $from $dest -}} {{- end -}} {{- else -}} {{- $target := path.Clean (path.Join (path.Dir $from) $u.Path) -}} {{- $frag := cond (ne $u.Fragment "") (printf "#%s" $u.Fragment) "" -}} {{- $page := false -}} {{- /* Each mounted folder and the content path it is mounted at (site/hugo.toml). */ -}} {{- range slice (slice "docs/" "docs/") (slice "site/content/" "") (slice "site/.gen/content/" "docs/") -}} {{- $dir := index . 0 -}} {{- if and (not $page) (or (hasPrefix $target $dir) (eq (printf "%s/" $target) $dir)) -}} {{- $page = site.GetPage (printf "/%s%s" (index . 1) (strings.TrimPrefix $dir (printf "%s/" $target) | strings.TrimSuffix "/")) -}} {{- end -}} {{- end -}} {{- if $page -}} {{- $href = printf "%s%s" $page.RelPermalink $frag -}} {{- else if fileExists $target -}} {{- $kind := cond (os.Stat $target).IsDir "tree" "blob" -}} {{- $href = printf "%s/%s/%s/%s%s" site.Params.repoURL $kind site.Params.commit $target $frag -}} {{- else -}} {{- errorf "%s: broken link %q (resolves to %s, which does not exist)" $from $dest $target -}} {{- end -}} {{- end -}} {{
